import hashlib
import json
import math
import os
import re


def embed(text):
    vector = [0.0] * 64
    for token in re.findall(r"\w+", text.casefold()):
        digest = hashlib.sha256(token.encode()).digest()
        vector[int.from_bytes(digest[:4], "big") % 64] += 1.0
    norm = math.sqrt(sum(value * value for value in vector)) or 1.0
    return [value / norm for value in vector]


def tokenizer(text):
    return [int.from_bytes(hashlib.sha256(token.encode()).digest()[:4], "big")
            for token in re.findall(r"\w+", text)]


def cases():
    with open(os.environ["OPENRESEARCH_INPUT"], encoding="utf-8") as stream:
        return json.load(stream)


def documents(inputs):
    value = inputs["documents"]
    if isinstance(value, str):
        value = json.loads(value)
    if not isinstance(value, list) or not value:
        raise ValueError("documents must be a nonempty list of {id,text}")
    identifiers = set()
    for document in value:
        if (not isinstance(document, dict) or not isinstance(document.get("id"), str)
                or not isinstance(document.get("text"), str) or document["id"] in identifiers):
            raise ValueError("invalid or duplicate document")
        identifiers.add(document["id"])
    return value


def write(predictions):
    with open(os.environ["OPENRESEARCH_PREDICTIONS"], "w", encoding="utf-8") as stream:
        json.dump(predictions, stream)
    print(json.dumps({"actual_predictions": len(predictions)}))
